\chapter{Exemple Sérieux : Théorie des groupes}

\section{Généralités}

Dans cette section, nous introduisons les éléments principaux pour comprendre la notion de groupe et
de morphismes de groupes.

\begin{definition}{(Groupe)}
  On appelle groupe tout quadruplet $(G,\cdot, e, \cdot^{-1})$ tel que G est un ensemble et 
  $\cdot$ une loi interne associative de neutre $e$, c'est-à-dire : 

  \begin{enumerate}
  \item $\forall x,y,z \in G, (x \cdot y) \cdot z = x \cdot ( y \cdot z)$.
  \item $\forall x \in G, e \cdot x = x \cdot e = x$.
  \end{enumerate}

  De plus, $\cdot^{-1} : G \to G$ vérifie :\[
    \forall x \in G, x \cdot x^{-1} = x^{-1} \cdot x = e
  \]
\end{definition}

\begin{remark}
  Dans la suite, si le contexte le permet, on notera simplement $G$ un groupe, et on notera le produit
  avec le symbole invisible, à savoir $x \cdot y$ deviendra $xy$.
  Dans presque tous les cas, les symboles $e$ et $\cdot^{-1}$ seront omis, c'est pourquoi on rencontrera
  souvent la notation $(G,\cdot)$ pour un groupe, ou encore $(G,+)$ pour un groupe lorsque celui-ci est
  commutatif.
\end{remark}

\begin{example}{}
  \blank
  \begin{enumerate}
    \item Pour $n \in \mathbb{N}$, $\GL_n (\mathbb{R})$ est un groupe.
    \item Pour $n \in \mathbb{N}^*, (\mathbb{Z}/n \mathbb{Z},+)$ est un groupe.
  \end{enumerate}
\end{example}

\begin{proposition}{}
  \label{prop:unicité inverse}
  Soit $G$ un groupe. Tout élément de $G$ admet un unique inverse pour la loi de composition interne.
\end{proposition}

\begin{proof}
    Soit $x \in G$, et soient $y, z \in G$ tels que $xy = yx =e$ et $xz = zx = e$.
    Montrons que $y=z$.

    On a : \begin{align*}
      yxz &= (yx)z = ez = z \\
          &= y(xz) = ye = y
    \end{align*}

    En mettant bout à bout les égalités, on obtient bien que $y=z$.
\end{proof}

\begin{definition}{}
  Soient $G,G'$ deux groupes et $f : G \to G'$ une application.
  On dit que $f$ est un morphisme de groupes si :
  \[
    \forall x,y \in G, f(xy) = f(x) f(y).
  \]
\end{definition}

\begin{remark}
  Si $f:G \to G'$ est un morphisme de groupe, alors $f(e_G) = e_{G'}$.
  Pour celles et ceux qui auraient l'habitude du formalisme des monoïdes,
  il n'est plus nécessaire de vérifier ce qu'il se passe pour l'élément neutre
  puisque la loi de groupe induit naturellement que le neutre est envoyé... sur le neutre !
\end{remark}

      \overline{i} & \mapsto \overline{ki}
    \end{cases}$ est un morphisme de groupes.
\end{exo}
